\section{Introduction}
\label{sec:intro}

\subsection{About}
\label{subsec:about}

This package provides \cs{clock} and \cs{calendardate}.
\cs{clock} prints the time of day.
\cs{calendardate} prints the calendar date.
These commands are ideal for text markup because their syntax is short,
flexible, easy-to-understand, and unambiguous.%
\footnote
  {%
    Not entirely unambiguous.
    See the key \key{priority} (\S\ref{subsec:date}).%
  }
The printed time of day and calendar date are highly customizable through a
variety of key-value options.
For advanced users, this package also provides robust \LaTeX3 programming
support.

\section{Commands}
\label{sec:cmds}

This section documents the commands provided by the \pkg{clockdate} package.

\begin{codedoc}{\clockdatesetup}
  \begin{syntax}
    \cs{clockdatesetup} \marg{key-value list}
  \end{syntax}
  Sets and processes the \pkg{clockdate} package keys (\S\ref{sec:keys}) in
  \meta{key-value list}.
  All assignments made by \cs{clockdatesetup} are local to the current group.
  Can be used mid-document.
\end{codedoc}

\begin{codedoc}{\clock}
  \begin{syntax}
    \cs{clock} \oarg{key-value list} \marg{clock(s)}
  \end{syntax}
  Prints the time of day specified by \meta{clock(s)}.
  \meta{clock(s)} must take any of the following forms:
  \begin{itemize}
    \item a single clock \meta{clock}
    \item a clock range \meta{clock_1}|--|\meta{clock_2}
    \item
    a comma-separated list where each item is a single clock or a clock range
  \end{itemize}

  \meta{clock} must take any of the following forms:
  \begin{itemize}
    \item \meta{hour}
    \item \meta{hour}|:|\meta{minute}
    \item \meta{hour}|:|\meta{minute}|:|\meta{second}
  \end{itemize}

  \meta{hour} must be an \meta{integer} from \num{0} to \num{24}.
  \meta{minute} must be an \meta{integer} from \num{0} to \num{59}.
  \meta{second} must be an \meta{integer} from \num{0} to \num{60}.
  (The value of \num{60} allows for leap seconds.)
  If $\meta{hour}=\num{24}$, then \meta{minute} and \meta{second} must be zero.

  The optional argument sets the \pkg{clockdate} package keys in
  \meta{key-value list} for only that instance of \cs{clock}.

  Additional parsing details for \meta{clock(s)}:
  \begin{itemize}
    \item |--| at the top level must only be used as a delimiter.
    \item
    \meta{hour}, \meta{minute}, and \meta{second} may be enclosed in braces.
    This prevents the parser from seeing the enclosed |--|.
    (|--| is valid input within an integer expression.)
    \item
    Colons of catcode 11 and 13 (active) at the top level are replaced by that
    of catcode 12.
    \item Active hyphens at the top level are replaced by that of catcode 12.
  \end{itemize}

  Table~\ref{table:clock} displays examples of \cs{clock}.
\end{codedoc}

\begin{codedoc}{\calendardate, \Calendardate}
  \begin{syntax}
    \cs{calendardate} \oarg{key-value list} \marg{date(s)}
  \end{syntax}
  Prints the calendar date specified by \meta{date(s)}.
  \meta{date(s)} must take any of the following forms:
  \begin{itemize}
    \item \meta{year}|-|\meta{month}|-|\meta{day(s)}
    \item \meta{month}|-|\meta{day(s)}
    \item \meta{year}|-|\meta{month(s)}
    \item \meta{year(s)}
    \item \meta{blank}|-|\meta{year}|-|\meta{month}|-|\meta{day(s)}
    \item \meta{blank}|-|\meta{year}|-|\meta{month(s)}
    \item \meta{blank}|-|\meta{year(s)}
  \end{itemize}

  If the first field is \meta{blank}, then \meta{year} is always negative
  (before the common era); if printing more than one year, this applies only to
  the very first \meta{year}.

  \meta{day(s)} must take any of the following forms:
  \begin{itemize}
    \item a single day \meta{day}
    \item a day range \meta{day_1}|--|\meta{day_2}
    \item a comma-separated list where each item is a single day or a day range
  \end{itemize}

  \meta{month(s)} must take any of the following forms:
  \begin{itemize}
    \item a single month \meta{month}
    \item a month range \meta{month_1}|--|\meta{month_2}
    \item
    a comma-separated list where each item is a single month or a month range
  \end{itemize}

  \meta{year(s)} must take any of the following forms:
  \begin{itemize}
    \item a single year \meta{year}
    \item a year range \meta{year_1}|--|\meta{year_2}
    \item
    a comma-separated list where each item is a single year or a year range
  \end{itemize}

  \meta{year} and \meta{day} must be an \meta{integer}.
  \meta{month} must be an \meta{integer} from \num{1} to \num{12}.
  Each calendar date must be valid.
  \date{2-29} is always valid if \meta{year} is omitted.
  If \meta{year} is provided, \date{2-29} is error-checked using the Julian
  calendar.

  The optional argument sets the \pkg{clockdate} package keys in
  \meta{key-value list} for only that instance of \cs{calendardate}.
  \cs{Calendardate} is equivalent to \cs{calendardate}, except that it always
  sets the key \key{capitalize-first=true} (\S\ref{subsec:date}) for only that
  instance of \cs{Calendardate}.

  Additional parsing details for \meta{date(s)}:
  \begin{itemize}
    \item |-| and |--| at the top level must only be used as a delimiter.
    \item
    \meta{year}, \meta{month}, and \meta{day} may be enclosed in braces.
    This prevents the parser from seeing the enclosed |-| or |--|.
    (|--| is valid input within an integer expression.)
    \item Active hyphens at the top level are replaced by that of catcode 12.
  \end{itemize}

  Table~\ref{table:date} displays examples of \cs{calendardate}.
\end{codedoc}

\section{Keys}
\label{sec:keys}

This section documents the keys provided by the \pkg{clockdate} package.
Set the package keys using \cs{clockdatesetup} (\S\ref{sec:cmds}).

\subsection{Preamble-only keys}
\label{subsec:preamblekeys}

\begin{codedoc}{define-hour-minute, \hour, \minute}
  At the \hook{begindocument} hook:
  \begin{itemize}
    \item
    Defines \cs{hour} and \cs{minute} as count registers that store the
    current hour and minute, even if \cs{hour} or \cs{minute} are already
    defined.
    \item Issues a warning if \cs{hour} or \cs{minute} are already defined.
  \end{itemize}
  This key can only be set in the preamble.
  \begin{codedemo}
    \clock{\hour}\\
    \clock{\hour:\minute}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{redefine-date, \date, \Date}
  At the \hook{begindocument} hook:
  \begin{itemize}
    \item Redefines \cs{date} as equivalent to \cs{calendardate}.
    \item
    Defines \cs{Date} as equivalent to \cs{Calendardate}, even if \cs{Date} is
    already defined.
    \item Issues a warning if \cs{Date} is already defined.
  \end{itemize}
  This key can only be set in the preamble.
  \begin{codedemo}
    \calendardate{1920-8-26}\\
    \calendardate{8-26}\\
    \calendardate{1920-8}\\
    \date{1920-8-26}\\
    \date{8-26}\\
    \date{1920-8}
  \end{codedemo}
\end{codedoc}

\subsection{\cs{clock}}
\label{subsec:clock}

This subsection documents the keys that modify the behavior of \cs{clock}
(\S\ref{sec:cmds}).

\begin{codedoc}{clock}
  \begin{syntax}
    \key{clock=12|24}
  \end{syntax}
  Choice key that sets whether the hour is printed using a 12-hour or 24-hour
  clock.
  If the value is |24|, the am/pm is always omitted.
  The initial value is |12|.
  \begin{codedemo}
    \clock{5:45}\\
    \clock{14:30}\\
    \clock{18:05--21:00}\\
    \clockdatesetup{clock=24}%
    \clock{5:45}\\
    \clock{14:30}\\
    \clock{18:05--21:00}
  \end{codedemo}
\end{codedoc}

\subsubsection{hour, minute, second}
\label{subsubsec:hms}

\begin{codedoc}{hour}
  \begin{syntax}
    \key{hour=h|hh}
  \end{syntax}
  Choice key that sets the format in which the hour is printed.
  |h| prints the hour with the minimum number of digits.
  |hh| prints the hour with two digits, adding a leading zero if necessary.
  The initial value is |h|.
  \begin{codedemo}
    \clock{4}\\
    \clock{9:25}\\
    \clockdatesetup{hour=hh}%
    \clock{4}\\
    \clock{9:25}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{minute}
  \begin{syntax}
    \key{minute=m|mm}
  \end{syntax}
  Choice key that sets the format in which the minute is printed.
  |m| prints the minute with the minimum number of digits.
  |mm| prints the minute with two digits, adding a leading zero if necessary.
  The initial value is |mm|.
  \begin{codedemo}
    \clock{9:00}\\
    \clock{13:05}\\
    \clockdatesetup{minute=m}%
    \clock{9:00}\\
    \clock{13:05}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{second}
  \begin{syntax}
    \key{second=s|ss}
  \end{syntax}
  Choice key that sets the format in which the second is printed.
  |s| prints the second with the minimum number of digits.
  |ss| prints the second with two digits, adding a leading zero if necessary.
  The initial value is |ss|.
  \begin{codedemo}
    \clock{16:00:00}\\
    \clock{9:23:03--9:23:09}\\
    \clockdatesetup{second=s}%
    \clock{16:00:00}\\
    \clock{9:23:03--9:23:09}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{hour-minute-separator}
  \begin{syntax}
    \key{hour-minute-separator=\meta{token list}}
  \end{syntax}
  Sets the separator between \meta{hour} and \meta{minute} to
  \meta{token list}.
  The initial value is |:|.
  \begin{codedemo}
    \clock{10:30}\\
    \clock{17:15--19:45}\\
    \clockdatesetup{hour-minute-separator=.}%
    \clock{10:30}\\
    \clock{17:15--19:45}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{minute-second-separator}
  \begin{syntax}
    \key{minute-second-separator=\meta{token list}}
  \end{syntax}
  Sets the separator between \meta{minute} and \meta{second} to
  \meta{token list}.
  The initial value is |:|.
  \begin{codedemo}
    \clock{7:34:15}\\
    \clock{10:00:09--12:15:35}\\
    \clockdatesetup{minute-second-separator=.}%
    \clock{7:34:15}\\
    \clock{10:00:09--12:15:35}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{time-separator}
  \begin{syntax}
    \key{time-separator=\meta{token list}}
  \end{syntax}
  Meta key that sets the value of the keys \key{hour-minute-separator} and
  \key{minute-second-separator} to \meta{token list}.
\end{codedoc}

\subsubsection{am/pm}
\label{subsubsec:ampm}

\begin{codedoc}{ampm}
  \begin{syntax}
    \key{ampm=\meta{choice}}
  \end{syntax}
  Choice key that sets the format in which the am/pm is printed.
  \meta{choice} must match any of the following:
  \begin{itemize}
    \item |A.M.|, |AM|, |A|
    \item |a.m.|, |am|, |a|
    \item |a.m.-scshape|, |am-scshape|, |a-scshape|
    \item |none|
  \end{itemize}
  The initial value is |AM|.

  The am/pm is omitted if \key{ampm=none} or \key{clock=24}
  (\S\ref{subsec:clock}).

  If \cs{clock} ends with an am/pm that ends with a period, then the trailing
  am/pm will not print an extra period if the next token after \cs{clock} is a
  period.
  If the am/pm prints the period, it is followed by \cs{@} because the sentence
  does not end here.
  \begin{codedemo}
    \clock[ampm=A.M.]{9--15}\\
    \clock[ampm=AM]{9--15}\\
    \clock[ampm=A]{9--15}\\
    \clock[ampm=a.m.]{9--15}\\
    \clock[ampm=am]{9--15}\\
    \clock[ampm=a]{9--15}\\
    \clock[ampm=a.m.-scshape]{9--15}\\
    \clock[ampm=am-scshape]{9--15}\\
    \clock[ampm=a-scshape]{9--15}\\
    \clock[ampm=none]{9--15}\\
    \clock[ampm=A.M.]{8}.\\
    \clock[ampm=A.M.]{16}.\\
    \clock[ampm=a.m.]{8}.\\
    \clock[ampm=a.m.]{16}.\\
    \clock[ampm=a.m.-scshape]{8}.\\
    \clock[ampm=a.m.-scshape]{16}.
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{ampm-separator}
  \begin{syntax}
    \key{ampm-separator=\meta{token list}}
  \end{syntax}
  Sets the separator before the am/pm to \meta{token list}.
  The initial value is \cs{nobreakspace}.
  \begin{codedemo}
    \clock{5:45}\\
    \clock{15:10}\\
    \clockdatesetup{ampm-separator=}%
    \clock{5:45}\\
    \clock{15:10}\\
    \clockdatesetup{ampm-separator=\thinspace}%
    \clock{5:45}\\
    \clock{15:10}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{ampm-omit-range}
  \begin{syntax}
    \key{ampm-omit-range=true|false}
  \end{syntax}
  Boolean key that sets if \meta{clock_1} omits the am/pm in a clock range if
  \meta{clock_1} and \meta{clock_2} would print the same meridiem and
  $\meta{clock_2}>\meta{clock_1}$.
  The initial value is |true|.
  \begin{codedemo}
    \clock{4--6}\\
    \clock{7,13--16:25}\\
    \clockdatesetup{ampm-omit-range=false}%
    \clock{4--6}\\
    \clock{7,13--16:25}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{ampm-omit-list}
  \begin{syntax}
    \key{ampm-omit-list=true|false}
  \end{syntax}
  Boolean key that sets if each \meta{clock}---except the very last---omits the
  am/pm if each \meta{clock} would print the same meridiem.
  The initial value is |true|.
  \begin{codedemo}
    \clock{5,7,9}\\
    \clock{6,8--11}\\
    \clock{18--19}\\
    \clockdatesetup{ampm-omit-list=false}%
    \clock{5,7,9}\\
    \clock{6,8--11}\\
    \clock{18--19}\\
    \clockdatesetup{ampm-omit-range=false}%
    \clock{5,7,9}\\
    \clock{6,8--11}\\
    \clock{18--19}
  \end{codedemo}
\end{codedoc}

\subsubsection{clocks}
\label{subsubsec:clocks}

\begin{codedoc}{clocks-pair-separator}
  \begin{syntax}
    \key{clocks-pair-separator=\meta{token list}}
  \end{syntax}
  Sets the separator between each item in \meta{clocks} to \meta{token list} if
  \meta{clocks} contains exactly two items.
  The initial value is | and |.
  \begin{codedemo}
    \clock{5,17}\\
    \clock[clocks-pair-separator={ or }]{5,17}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{clocks-nonfinal-separator}
  \begin{syntax}
    \key{clocks-nonfinal-separator=\meta{token list}}
  \end{syntax}
  Sets the separator between each item in \meta{clocks}---except the last
  two---to \meta{token list} if \meta{clocks} contains three or more items.
  The initial value is |, |.
  \begin{codedemo}
    \clock{11,12:30,2:45:00}\\
    \clock[clocks-nonfinal-separator={; }]{11,12:30,2:45:00}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{clocks-final-separator}
  \begin{syntax}
    \key{clocks-final-separator=\meta{token list}}
  \end{syntax}
  Sets the separator between the last two items in \meta{clocks} to
  \meta{token list} if \meta{clocks} contains three or more items.
  The initial value is |, and |.
  \begin{codedemo}
    \clock{11,12:30,2:45:00}\\
    \clock[clocks-final-separator={ and }]{11,12:30,2:45:00}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{clocks-separator}
  \begin{syntax}
    \key{clocks-separator=\meta{token list}}
  \end{syntax}
  Meta key that sets the value of the following keys to \meta{token list}:
  \begin{itemize}
    \item \key{clocks-pair-separator}
    \item \key{clocks-nonfinal-separator}
    \item \key{clocks-final-separator}
  \end{itemize}
\end{codedoc}

\begin{codedoc}{clocks-range-separator}
  \begin{syntax}
    \key{clocks-range-separator=\meta{token list}}
  \end{syntax}
  Sets the separator between \meta{clock_1} and \meta{clock_2} in a clock range
  to \meta{token list}.
  The initial value is | to |.
  \begin{codedemo}
    \clock{6--8:30}\\
    \clock{11:20--14:10}\\
    \clockdatesetup{clocks-range-separator=\textendash}%
    \clock{6--8:30}\\
    \clock{11:20--14:10}
  \end{codedemo}
\end{codedoc}

\subsection{\cs{calendardate}}
\label{subsec:date}

This subsection documents the keys that modify the behavior of
\cs{calendardate} (\S\ref{sec:cmds}).

\begin{codedoc}{priority}
  \begin{syntax}
    \key{priority=MD|YM}
  \end{syntax}
  Choice key that sets the typeset representation if the parser cannot
  distinguish between \meta{month}|-|\meta{day(s)} and
  \meta{year}|-|\meta{month(s)}.
  This situation occurs if \meta{date(s)} contains exactly two fields and all
  values within are an \meta{integer} from \num{1} to \num{12}.
  The initial value is |MD|.
  \begin{codedemo}
    \calendardate{6-7}\\
    \calendardate{12-1,2,3}\\
    \clockdatesetup{priority=YM}%
    \calendardate{6-7}\\
    \calendardate{12-1,2,3}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{capitalize-first}
  \begin{syntax}
    \key{capitalize-first=true|false}
  \end{syntax}
  Boolean key that sets if the month will be capitalized if it is printed at
  the start of \cs{calendardate}.
  The initial value is |false|.
  See \cs{Calendardate} (\S\ref{sec:cmds}).
  See \S\ref{subsec:expl3date} for more information.
\end{codedoc}

\begin{codedoc}{order}
  \begin{syntax}
    \key{order=DMY|YMD|MDY}
  \end{syntax}
  Choice key that sets the order in which the year, month, and day are printed.
  The initial value is |DMY|.
  \begin{codedemo}
    \calendardate{1929-10-24}\\
    \calendardate{10-31}\\
    \calendardate{2023-5}\\
    \clockdatesetup{order=YMD}%
    \calendardate{1929-10-24}\\
    \calendardate{10-31}\\
    \calendardate{2023-5}\\
    \clockdatesetup{order=MDY}%
    \calendardate{1929-10-24}\\
    \calendardate{10-31}\\
    \calendardate{2023-5}
  \end{codedemo}
\end{codedoc}

\subsubsection{year, month, day}
\label{subsubsec:ymd}

\begin{codedoc}{year}
  \begin{syntax}
    \key{year=y|yy|yyyy}
  \end{syntax}
  Choice key that sets the format in which the year is printed.
  |y| prints the year with the minimum number of digits.
  |yy| prints the year with only the last two digits, adding a leading zero if
  the year is a single digit.
  |yyyy| prints the year with a minimum of four digits, adding leading zeroes
  if necessary.
  The initial value is |y|.
  \begin{codedemo}
    \begin{multicols}{3}
      \calendardate{2-12-25}\\
      \calendardate{19-12-25}\\
      \calendardate{199-12-25}\\
      \calendardate{1999-12-25}\\
      \calendardate{-1-12-25}\\
      \calendardate{-18-12-25}\\
      \calendardate{-198-12-25}\\
      \calendardate{-1998-12-25}\\
      \clockdatesetup{year=yy}%
      \calendardate{2-12-25}\\
      \calendardate{19-12-25}\\
      \calendardate{199-12-25}\\
      \calendardate{1999-12-25}\\
      \calendardate{-1-12-25}\\
      \calendardate{-18-12-25}\\
      \calendardate{-198-12-25}\\
      \calendardate{-1998-12-25}\\
      \clockdatesetup{year=yyyy}%
      \calendardate{2-12-25}\\
      \calendardate{19-12-25}\\
      \calendardate{199-12-25}\\
      \calendardate{1999-12-25}\\
      \calendardate{-1-12-25}\\
      \calendardate{-18-12-25}\\
      \calendardate{-198-12-25}\\
      \calendardate{-1998-12-25}
    \end{multicols}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{yy-year-prefix}
  \begin{syntax}
    \key{yy-year-prefix=\meta{token list}}
  \end{syntax}
  Prepends \meta{token list} to the printed year if \key{year=yy} and the year
  omits 1 or more digits.
  The initial value is \cs{rq}.
  \begin{codedemo}
    \clockdatesetup{year=yy}%
    \calendardate{2003-3-20}\\
    \calendardate{732-10-10}\\
    \clockdatesetup{yy-year-prefix=}%
    \calendardate{2003-3-20}\\
    \calendardate{732-10-10}
  \end{codedemo}
\end{codedoc}

\begin{codedoc}{month}
  \begin{syntax}
    \key{month=m|mm|mmm|mmmm|abbr}
  \end{syntax}
  Choice key that sets the format in which the month is printed.
  |m| prints the month with the minimum number of digits.
  |mm| prints the two-digit month, adding a leading zero if necessary.
  |mmm| prints the 3-letter month.
  |mmmm| prints the month with the name of the month fully spelled out.
  |abbr| prints the abbreviated month.
  The initial value is |mmmm|.

  By default, |abbr| prints the month in the same form as |mmmm|, except that
  some months are abbreviated: Jan., Feb., Aug., Sept., Oct., Nov., Dec.
  If \cs{calendardate} ends with a month that ends with a period, then the
  trailing month will not print an extra period if the next token after
  \cs{calendardate} is a period.

    \calendardate[month=mm]{1-1}\\

\subsubsection{era}
\label{subsubsec:era}

\begin{codedoc}{era}
  \begin{syntax}
    \key{era=\meta{choice}}
  \end{syntax}
  Choice key that sets the format in which the era is printed.
  \meta{choice} must match any of the following:
  \begin{itemize}
    \item |C.E.|, |CE|, |c.e.-scshape|, |ce-scshape|
    \item |A.D.|, |AD|, |a.d.-scshape|, |ad-scshape|
    \item |none|
    \item |integer|
  \end{itemize}
  The initial value is |CE|.

  |none| omits the era.
  |integer| omits the era and instead prints the year as an integer which may
  be positive, zero, or negative.

  If \cs{calendardate} ends with an era that ends with a period, then the
  trailing era will not print an extra period if the next token after
  \cs{calendardate} is a period.
  If the era prints the period, it is followed by \cs{@} because the sentence
  does not end here.
  \begin{codedemo}
    \calendardate[era=C.E.]{-515--70}\\
    \calendardate[era=CE]{-515--70}\\
    \calendardate[era=c.e.-scshape]{-515--70}\\
    \calendardate[era=ce-scshape]{-515--70}\\
    \calendardate[era=A.D.]{-515--70}\\
    \calendardate[era=AD]{-515--70}\\
    \calendardate[era=a.d.-scshape]{-515--70}\\
    \calendardate[era=ad-scshape]{-515--70}\\
    \calendardate[era=none]{-515--70}\\
    \calendardate[era=integer]{-515--70}

\subsection{Loading precompiled keys}
\label{subsec:loadkeys}

\begin{codedoc}{reset}
  Meta key that sets all the \pkg{clockdate} keys to their initial values.
\end{codedoc}

\begin{codedoc}{load, \l_clockdate_load_prop}
  \begin{syntax}
    \key{load=\meta{string}}
  \end{syntax}
  The key \key{load} should be used to load precompiled \pkg{clockdate} keys.
  \meta{string} must be a \meta{key} in \cs{l_clockdate_load_prop}.
  The \meta{value} of each \meta{key} in \cs{l_clockdate_load_prop} should
  contain precompiled \pkg{clockdate} keys.
  \cs{l_clockdate_load_prop} initially contains no entries.

  The following example initializes \key{load=or} to load precompiled keys:
  \begin{verbatim}
    \ExplSyntaxOn
    \keys_precompile:nnN { clockdate }
      {
          clocks-pair-separator  = {  ~or~ }
        , clocks-final-separator = { ,~or~ }
        , days-pair-separator    = {  ~or~ }
        , days-final-separator   = { ,~or~ }
        , months-pair-separator  = {  ~or~ }
        , months-final-separator = { ,~or~ }
        , years-pair-separator   = {  ~or~ }
        , years-final-separator  = { ,~or~ }
      }
      \l_tmpa_tl
    \prop_put:NnV \l_clockdate_load_prop { or } \l_tmpa_tl
    \ExplSyntaxOff
  \end{verbatim}
\end{codedoc}

\subsection{The extra period problem}
\label{subsec:extraperiod}

\begin{codedoc}{\clockdate_end_period:}
  Function used in \cs{clock} and \cs{calendardate} to avoid mistakenly
  printing two consecutive periods if the next token after \cs{clock} or
  \cs{calendardate} is a period.

  This function is used in:
  \begin{itemize}
    \item \cs{clockdate_print_am:}, \cs{clockdate_print_pm:}
    \item \cs{clockdate_print_month:n}, \cs{l_clockdate_months_abbr_seq}
    \item \cs{clockdate_print_year:n}
  \end{itemize}

  Prints a period unless both of the following are true:
  \begin{itemize}
    \item
    This function is used at the very end of \cs{clock} or \cs{calendardate},
    followed by zero or more tokens with the same meaning as \cs{group_end:}.
    \item The next token after \cs{clock} or \cs{calendardate} is a period.
  \end{itemize}

  The gobbling and replacement of \cs{group_end:} tokens or equivalent ensures
  that the period has the same formatting as the preceding text.
  If \cs{clockdate_end_period:} prints the period, it is followed by \cs{@}
  because the sentence does not end here.
\end{codedoc}
