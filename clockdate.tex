\section{Introduction}
\label{sec:intro}

\subsection{About}
\label{subsec:about}

This package provides \cs{clock} and \cs{calendardate}.
\cs{clock} prints the time of day.
\cs{calendardate} prints the calendar date.
These commands are ideal for text markup because their syntax is short,
flexible, easy-to-understand, and unambiguous.%
\footnote
  {%
    Not entirely unambiguous.
    See the key \key{priority} (\S\ref{subsec:date}).%
  }
The printed time of day and calendar date are highly customizable through a
variety of key-value options.
For advanced users, this package also provides robust \LaTeX3 programming
support.

\section{Commands}
\label{sec:cmds}

This section documents the commands provided by the \pkg{clockdate} package.

\begin{code}{\clockdatesetup}
  \begin{syntax}
    \cs{clockdatesetup} \marg{key-value list}
  \end{syntax}
  Sets and processes the \pkg{clockdate} package keys (\S\ref{sec:keys}) in
  \meta{key-value list}.
  All assignments made by \cs{clockdatesetup} are local to the current group.
  Can be used mid-document.
\end{code}

\begin{code}{\clock}
  \begin{syntax}
    \cs{clock} \oarg{key-value list} \marg{clock(s)}
  \end{syntax}
  Prints the time of day specified by \meta{clock(s)}.
  \meta{clock(s)} must take any of the following forms:
  \begin{itemize}
    \item a single clock \meta{clock}
    \item a clock range \meta{clock_1}|--|\meta{clock_2}
    \item
    a comma-separated list where each item is a single clock or a clock range
  \end{itemize}

  \meta{clock} must take any of the following forms:
  \begin{itemize}
    \item \meta{hour}
    \item \meta{hour}|:|\meta{minute}
    \item \meta{hour}|:|\meta{minute}|:|\meta{second}
  \end{itemize}

  \meta{hour} must be an \meta{integer} from \num{0} to \num{24}.
  \meta{minute} must be an \meta{integer} from \num{0} to \num{59}.
  \meta{second} must be an \meta{integer} from \num{0} to \num{60}.
  (The value of \num{60} allows for leap seconds.)
  If $\meta{hour}=\num{24}$, then \meta{minute} and \meta{second} must be zero.

  The optional argument sets the \pkg{clockdate} package keys in
  \meta{key-value list} for only that instance of \cs{clock}.

  Additional parsing details for \meta{clock(s)}:
  \begin{itemize}
    \item
    \meta{hour}, \meta{minute}, and \meta{second} may be enclosed in braces.
    This prevents the parser from seeing the enclosed |--|.
    (|--| is valid input within an integer expression.)
    \item
    Colons of catcode 11 and 13 (active) at the top level are replaced by that
    of catcode 12.
    \item Active hyphens at the top level are replaced by that of catcode 12.
  \end{itemize}

  Table~\ref{table:clock} displays examples of \cs{clock} with many different
  inputs.
\end{code}

\input{clockdate-input-clock.tex}

\begin{code}{\calendardate, \Calendardate}
  \begin{syntax}
    \cs{calendardate} \oarg{key-value list} \marg{date(s)}
  \end{syntax}
  Prints the calendar date specified by \meta{date(s)}.
  \meta{date(s)} must take any of the following forms:
  \begin{itemize}
    \item \meta{year}|-|\meta{month}|-|\meta{day(s)}
    \item \meta{month}|-|\meta{day(s)}
    \item \meta{year}|-|\meta{month(s)}
    \item \meta{year}
  \end{itemize}

  \meta{year} may be substituted with \meta{empty}|-|\meta{year} to ensure that
  the year is always before the common era.

  \meta{day(s)} must take any of the following forms:
  \begin{itemize}
    \item a single day \meta{day}
    \item a day range \meta{day_1}|--|\meta{day_2}
    \item a comma-separated list where each item is a single day or a day range
  \end{itemize}

  \meta{month(s)} must take any of the following forms:
  \begin{itemize}
    \item a single month \meta{month}
    \item a month range \meta{month_1}|--|\meta{month_2}
    \item
    a comma-separated list where each item is a single month or a month range
  \end{itemize}

  \meta{year} and \meta{day} must be an \meta{integer}.
  \meta{month} must be an \meta{integer} from \num{1} to \num{12}.
  Each calendar date must be valid.
  \date{2-29} is always valid if \meta{year} is omitted.
  If \meta{year} is provided, \date{2-29} is error-checked using the Julian
  calendar.

  The optional argument sets the \pkg{clockdate} package keys in
  \meta{key-value list} for only that instance of \cs{calendardate}.
  \cs{Calendardate} is equivalent to \cs{calendardate}, except that it always
  sets the key \key{capitalize-first=true} (\S\ref{subsec:date}) for only that
  instance of \cs{Calendardate}.

  Additional parsing details for \meta{date(s)}:
  \begin{itemize}
    \item |-| and |--| at the top level must only be used as a delimiter.
    \item
    \meta{year}, \meta{month}, and \meta{day} may be enclosed in braces.
    This prevents the parser from seeing the enclosed |-| or |--|.
    (|--| is valid input within an integer expression.)
    \item Active hyphens at the top level are replaced by that of catcode 12.
  \end{itemize}

  Table~\ref{table:date} displays examples of \cs{calendardate} with many
  different inputs.
\end{code}

\section{Keys}
\label{sec:keys}

This section documents the keys provided by the \pkg{clockdate} package.
Set the package keys using \cs{clockdatesetup} (\S\ref{sec:cmds}).

\begin{code}{define-current-time, \hour, \minute}
  Defines \cs{hour} and \cs{minute} as count registers that store the current
  hour and minute, even if \cs{hour} or \cs{minute} have already been defined.
  This is done at the \hook{begindocument} hook.
  This key can only be used in the preamble.
\end{code}

\begin{code}{redefine-date, \date, \Date}
  Redefines \cs{date} as equivalent to \cs{calendardate}.
  Defines \cs{Date} as equivalent to \cs{Calendardate}.
  This is done at the \hook{begindocument} hook so that the original
  definition of \cs{date} remains usuable in the preamble.
  This key can only be used in the preamble.
\end{code}

\begin{code}{reset}
  Meta key that sets all the \pkg{clockdate} package keys to their initial
  values.
\end{code}

\begin{code}{load}
  \begin{syntax}
    \key{load=\meta{string}}
  \end{syntax}
  Used to load precompiled \pkg{clockdate} keys.
  \meta{string} must be a key in \cs{l_clockdate_load_prop}
  (\S\ref{subsec:save}).
\end{code}

\subsection{\cs{clock}}
\label{subsec:clock}

This subsection documents the keys that modify the behavior of \cs{clock}
(\S\ref{sec:cmds}).

\begin{code}{clock}
  \begin{syntax}
    \key{clock=12|24}
  \end{syntax}
  Choice key that sets whether the hour is printed using a 12-hour or 24-hour
  clock.
  If the value is |24|, the am/pm is always omitted.
  The initial value is |12|.
\end{code}

\subsubsection{Printing the hour, minute, and second}
\label{subsubsec:printhms}

\begin{code}{hour}
  \begin{syntax}
    \key{hour=h|hh}
  \end{syntax}
  Choice key that sets the format in which the hour is printed.
  |h| prints the hour with the minimum number of digits.
  |hh| prints the hour with two digits, adding a leading zero if necessary.
  The initial value is |h|.
\end{code}

\begin{code}{minute}

\begin{code}{second}

\subsubsection{Printing the am/pm}
\label{subsubsec:printampm}

\begin{code}{ampm}
  \begin{syntax}
    \key{ampm=a.m.|am|a|A.M.|AM|A|none}
  \end{syntax}
  Choice key that sets the format in which the am/pm is printed.
  The initial value is |AM|.
  \input{clockdate-input-ampm.tex}

  The am/pm is omitted if \key{ampm=none} or \key{clock=24}.

  If the value is |a.m.| or |A.M.|, the trailing am/pm will not print an extra
  period if the next token after \cs{clock} is a period.
  If the am/pm prints the period, it is followed by \cs{@} because the sentence
  does not end here.
\end{code}

\begin{code}{ampm-separator}
  \begin{syntax}
    \key{ampm-separator=\meta{token list}}
  \end{syntax}
  Sets the separator before the am/pm to \meta{token list}.

\end{code}

\begin{code}{ampm-omit-range}
  \begin{syntax}
    \key{ampm-omit-range=true|false}
  \end{syntax}
  Boolean key that sets if \meta{clock_1} omits the am/pm in a clock range if
  \meta{clock_1} and \meta{clock_2} would print the same meridiem and
  $\meta{clock_2}>\meta{clock_1}$.

\end{code}

\begin{code}{ampm-omit-list}
  \begin{syntax}
    \key{ampm-omit-list=true|false}
  \end{syntax}
  Boolean key that sets if each \meta{clock}---except the very last---omits the
  am/pm if each \meta{clock} would print the same meridiem.
  The initial value is |true|.
\end{code}

\subsubsection{Printing the year, month, and day}
\label{subsubsec:printymd}

\begin{code}{year}
  \begin{syntax}
    \key{year=y|yy|yyyy}
  \end{syntax}
  Choice key that sets the format in which the year is printed.
  |y| prints the year with the minimum number of digits.
  |yy| prints the year with only the last two digits, adding a leading zero if
  the year is a single digit.
  |yyyy| prints the year with a minimum of four digits, adding leading zeroes
  if necessary.
  The initial value is |y|.
\end{code}

\begin{code}{month}
  \begin{syntax}
    \key{month=m|mm|mmm|mmmm}
  \end{syntax}
  Choice key that sets the format in which the month is printed.
  |m| prints the month with the minimum number of digits.
  |mm| prints the month with two digits, adding a leading zero if necessary.
  |mmm| prints the month in short form.
  |mmmm| prints the month with the name of the month fully spelled out.
  The initial value is |mmmm|.

  By default, |mmm| prints the month in the same form as |mmmm|, except that
  some months are abbreviated: Jan., Feb., Aug., Sept., Oct., Nov., Dec.
  The abbreviated month will not print an extra period if the next token after
  \cs{calendardate} is a period and \cs{calendardate} ends with an abbreviated
  month that ends in a period.
  If the abbreviated month prints the period, it is followed by \cs{@} because
  the sentence does not end here.
  See \S\ref{subsec:expl3date} for more details.
\end{code}

\begin{code}{day}
  \begin{syntax}
    \key{day=d|dd}
  \end{syntax}
  Choice key that sets the format in which the day is printed.
  |d| prints the day with the minimum number of digits.
  |dd| prints the day with two digits, adding a leading zero if necessary.
  The initial value is |d|.
\end{code}

\begin{code}{day-ordinal}
  \begin{syntax}
    \key{day-ordinal=true|false}
  \end{syntax}
  Boolean key that sets if the day is printed as an ordinal day (e.g.
  1\textsuperscript{st}, 4\textsuperscript{th}).
  The initial value is |false|.
  You can customize the display of the ordinal day by redefining

\end{code}

\subsection{Hooks}
\label{subsec:hooks}

This subsection documents the hooks provided by the \pkg{clockdate} package.

\begin{code}
  {
    cmd/clock/after,
    cmd/calendardate/after,
    cmd/Calendardate/after,
    cmd/date/after,
    cmd/Date/after
  }
  These generic |cmd| hooks are disabled because they would prevent the command
  from testing if the next token is a period (See \S\ref{subsubsec:printampm},
  \S\ref{subsubsec:printymd}, and \S\ref{subsubsec:printera}).
  \hook{cmd/date/after} and \hook{cmd/Date/after} are disabled only if the key
  \key{redefine-date} is set (\S\ref{sec:keys}).
\end{code}

\begin{code}{clockdate/clock}
  Hook used in \cs{clock} after parsing its input and setting all the variables
  in \S\ref{subsec:varclock}, but before printing the time of day.
  This hook should not be used to set the \pkg{clockdate} keys.
\end{code}

\begin{code}{clockdate/date}
  Hook used in \cs{calendardate} after parsing its input and setting all the
  variables in \S\ref{subsec:vardate}, but before printing the calendar date.
  This hook should not be used to set the \pkg{clockdate} keys.
\end{code}

\subsection{Precompiling the keys}
\label{subsec:save}

\begin{code}{\l_clockdate_load_prop}
  Property list used in the key \key{load} (\S\ref{sec:keys}).
  The value of the key \key{load} must be a key in \cs{l_clockdate_load_prop}.
  The value of each key in \cs{l_clockdate_load_prop} should contain
  precompiled \pkg{clockdate} keys.
  \cs{l_clockdate_load_prop} initially contains no entries.

  Example usage:
  \begin{verbatim}
\ExplSyntaxOn
\keys_precompile:nnN { clockdate }
  {
      clocks-pair-separator  = {  ~or~ }
    , clocks-final-separator = { ,~or~ }
    , days-pair-separator    = {  ~or~ }
    , days-final-separator   = { ,~or~ }
    , months-pair-separator  = {  ~or~ }
    , months-final-separator = { ,~or~ }
  }
  \l_tmpa_tl
\prop_put:NnV \l_clockdate_load_prop { or } \l_tmpa_tl
\ExplSyntaxOff
  \end{verbatim}
\end{code}

  \cs{calendardate}.

  The value is \meta{empty} if \meta{day(s)} is omitted.
\end{code}

\subsection{The extra period problem}
\label{subsec:extraperiod}

\begin{code}{\clockdate_end_period:}
  Function used in \cs{clock} and \cs{calendardate} to avoid mistakenly
  printing two consecutive periods if the next token after \cs{clock} or
  \cs{calendardate} is a period.

  This function is used in:
  \begin{itemize}
    \item \cs{clockdate_print_am:}, \cs{clockdate_print_pm:}
    \item \cs{clockdate_print_month:n}, \cs{l_clockdate_months_mmm_seq}
    \item \cs{clockdate_print_year:n}
  \end{itemize}

  Prints a period unless both of the following are true:
  \begin{itemize}
    \item
    This function is used at the very end of \cs{clock} or \cs{calendardate},
    followed by zero or more tokens with the same meaning as \cs{group_end:}.
    \item The next token after \cs{clock} or \cs{calendardate} is a period.
  \end{itemize}

  The gobbling and replacement of \cs{group_end:} tokens or equivalent ensures
  that the period has the same formatting as the preceding text.
  If \cs{clockdate_end_period:} prints the period, it is followed by \cs{@}
  because the sentence does not end here.
\end{code}
